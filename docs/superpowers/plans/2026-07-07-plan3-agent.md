- **Format-Gesetz importieren, nie kopieren:** `agent/*` importiert canon/bind/

- **Synth-uuid-Formel (Architektur-Entscheidung 1, verbindlich):**
  `record_uuid = "synth:" + sha256(utf8(f"{session_id}:{line_index}:{record_sha256}")).hexdigest()`
  Referenzvektor: session_id `aaaabbbb-cccc-dddd-eeee-ffff00001111`,
  line_index `7`, record_sha256
  `e4837f558dbf8eb95287c2fdaee928f8c3c617f5b9e21e27eeaff50bbdfe4002` ->
  `synth:264bdd15f80c08d2b17126f9cde43b771ea61cea63530164d3ed496e4baa703d`
- **line_index-Regel:** 1-basierte laufende Nummer der NICHT-LEEREN Zeilen einer
  Datei (Leerzeilen zählen nicht; nicht-parsebare nicht-leere Zeilen zählen,
  werden aber übersprungen und geloggt - dokumentierte v0-Limitation: raw muss
  ein JSON-Objekt pro Zeile sein). Nur vollständige Zeilen (mit `\n`) werden
  konsumiert; eine angerissene letzte Zeile bleibt für den nächsten Lauf.
- **session_id-Regel** (spiegelt die Engine-Semantik): Subagent-Dateien
  (`/subagents/` im Pfad) -> Datei-Stem; sonst `raw["sessionId"]` falls
  vorhanden, sonst Datei-Stem.
- **Sidecar-Regel (Entscheidung 3):** zu `agent-*.jsonl` gehörendes
  `agent-*.meta.json` wird als synthetischer Record verschickt:
  `raw = {"type": "x-meta-sidecar", "file": "<basename>", "content": <sidecar-json>}`,
  session_id = Stem der agent-Datei, line_index = 1. Erneut verschickt nur,
  wenn sich der Sidecar-Inhalt ändert (sha256-Tracking im State).
- **Rotation/Truncation:** inode-Wechsel ODER geschrumpfte Datei -> Offset und
  line_index auf 0 (Neulesen; Server-Dedup verhindert Doppelzählung).
- **Secrets:** Private Key als 32-Byte-Rohdatei, `chmod 0600`; Config-JSON
  ebenfalls 0600 (enthält Transport-Token). Beides niemals committen (liegt
  außerhalb des Repos unter `~/.config/audit-agent/` bzw. im Test unter tmp).
- Nach jedem Task: `.venv/bin/pytest tests/ -q` grün. Kein Docker-/Schema-
  Umbau in diesem Plan; die laufende `audit-collector-db` wird genutzt.

```python
"""Wrapper-Records bauen (Spec §4 + Architektur-Entscheidungen 1-3).

Importiert das Format-Gesetz aus collector.* - NIE kopieren.
"""
import hashlib

from collector.canon import record_sha256_hex
from collector.chain import bind_hex
from collector.crypto import sign_bind


def synth_record_uuid(session_id: str, line_index: int,
                      record_sha256: str) -> str:
    h = hashlib.sha256(
        f"{session_id}:{line_index}:{record_sha256}".encode("utf-8")
    ).hexdigest()
    return f"synth:{h}"


def extract_session_id(raw: dict, file_stem: str, is_subagent: bool) -> str:
    if is_subagent:
        return file_stem
    sid = raw.get("sessionId")
    return sid if isinstance(sid, str) and sid else file_stem


def build_wrapper(raw: dict, *, session_id: str, line_index: int,
                  captured_at: str, identity: dict, priv_raw: bytes) -> dict:
    rs = record_sha256_hex(raw)
    uuid = raw.get("uuid")
    record_uuid = (uuid if isinstance(uuid, str) and uuid
                   else synth_record_uuid(session_id, line_index, rs))
    parent = raw.get("parentUuid")
    b = bind_hex(identity["tenant"], session_id, record_uuid, rs)
    return {
        "agent_id": identity["agent_id"],
        "pubkey_id": identity["pubkey_id"],
        "tenant": identity["tenant"],
        "seat": {"machine": identity["machine"],
                 "os_user": identity["os_user"],
                 "source_label": identity["source_label"]},
        "session_id": session_id,
        "record_uuid": record_uuid,
        "parent_uuid": parent if isinstance(parent, str) and parent else None,
        "line_index": line_index,
        "captured_at": captured_at,
        "record_sha256": rs,
        "sig": sign_bind(priv_raw, b),
        "raw": raw,
    }


def build_sidecar_wrapper(sidecar_json: dict, *, file_name: str,
                          agent_file_stem: str, captured_at: str,
                          identity: dict, priv_raw: bytes) -> dict:
    raw = {"type": "x-meta-sidecar", "file": file_name,
           "content": sidecar_json}
    return build_wrapper(raw, session_id=agent_file_stem, line_index=1,
                         captured_at=captured_at, identity=identity,
                         priv_raw=priv_raw)
```

```python
"""Agent-Konfiguration + Schluesseldatei. Beides 0600, beides ausserhalb des Repos."""
import json
import os
from dataclasses import asdict, dataclass
from pathlib import Path

DEFAULT_CONFIG_PATH = Path(os.path.expanduser("~/.config/audit-agent/config.json"))


@dataclass
class AgentConfig:
    server_url: str
    tenant: str
    agent_id: str
    pubkey_id: str
    transport_token: str
    machine: str
    os_user: str
    source_label: str
    key_path: str
    state_path: str
    projects_dirs: list


def save_config(cfg: AgentConfig, path) -> None:
    p = Path(path)
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text(json.dumps(asdict(cfg), indent=2, ensure_ascii=False),
                 encoding="utf-8")
    os.chmod(p, 0o600)


def load_config(path) -> AgentConfig:
    return AgentConfig(**json.loads(Path(path).read_text(encoding="utf-8")))


def save_private_key(priv_raw: bytes, key_path) -> None:
    p = Path(key_path)
    p.parent.mkdir(parents=True, exist_ok=True)
    with open(p, "wb") as f:
        f.write(priv_raw)
    os.chmod(p, 0o600)


def load_private_key(key_path) -> bytes:
    data = Path(key_path).read_bytes()
    if len(data) != 32:
        raise ValueError(f"key file {key_path}: expected 32 bytes, got {len(data)}")
    return data


def identity(cfg: AgentConfig) -> dict:
    return {"agent_id": cfg.agent_id, "pubkey_id": cfg.pubkey_id,
            "tenant": cfg.tenant, "machine": cfg.machine,
            "os_user": cfg.os_user, "source_label": cfg.source_label}
```

```python
"""Lokaler Agent-State (Spec §8): SQLite, pro Datei inode/size/offset/line_index."""
import sqlite3

_SCHEMA = """
CREATE TABLE IF NOT EXISTS files (
  path       TEXT PRIMARY KEY,
  inode      INTEGER NOT NULL,
  size       INTEGER NOT NULL,
  offset     INTEGER NOT NULL,
  line_index INTEGER NOT NULL,
  updated_at TEXT NOT NULL DEFAULT (datetime('now'))
);
CREATE TABLE IF NOT EXISTS sidecars (
  path           TEXT PRIMARY KEY,
  content_sha256 TEXT NOT NULL,
  updated_at     TEXT NOT NULL DEFAULT (datetime('now'))
);
"""


def open_state(path) -> sqlite3.Connection:
    conn = sqlite3.connect(str(path))
    conn.executescript(_SCHEMA)
    return conn


def get_file_state(conn, path):
    row = conn.execute(
        "SELECT inode, size, offset, line_index FROM files WHERE path = ?",
        (str(path),)).fetchone()
    if row is None:
        return None
    return {"inode": row[0], "size": row[1], "offset": row[2],
            "line_index": row[3]}


def set_file_state(conn, path, inode, size, offset, line_index) -> None:
    conn.execute(
        "INSERT INTO files (path, inode, size, offset, line_index)"
        " VALUES (?,?,?,?,?)"
        " ON CONFLICT(path) DO UPDATE SET inode=excluded.inode,"
        " size=excluded.size, offset=excluded.offset,"
        " line_index=excluded.line_index, updated_at=datetime('now')",
        (str(path), inode, size, offset, line_index))
    conn.commit()


def sidecar_seen(conn, path, content_sha256: str) -> bool:
    row = conn.execute(
        "SELECT content_sha256 FROM sidecars WHERE path = ?",
        (str(path),)).fetchone()
    return row is not None and row[0] == content_sha256


def mark_sidecar(conn, path, content_sha256: str) -> None:
    conn.execute(
        "INSERT INTO sidecars (path, content_sha256) VALUES (?,?)"
        " ON CONFLICT(path) DO UPDATE SET content_sha256=excluded.content_sha256,"
        " updated_at=datetime('now')",
        (str(path), content_sha256))
    conn.commit()
```

```python
"""Inkrementeller Datei-Scan. Konsumiert nur vollstaendige Zeilen;
Rotation (inode-Wechsel) oder Truncation (geschrumpft) -> von vorn."""
import json
from pathlib import Path

from . import state as st


def discover_files(projects_dirs):
    for d in projects_dirs:
        base = Path(d).expanduser()
        if base.exists():
            yield from sorted(base.rglob("*.jsonl"))


def read_new_lines(conn, path):
    path = Path(path)
    stat_ = path.stat()
    prev = st.get_file_state(conn, path)
    offset, line_index = 0, 0
    if (prev and prev["inode"] == stat_.st_ino
            and prev["size"] <= stat_.st_size):
        offset, line_index = prev["offset"], prev["line_index"]
    with open(path, "rb") as f:
        f.seek(offset)
        buf = f.read()
    end = buf.rfind(b"\n")
    if end < 0:
        return [], offset, line_index
    parsed = []
    for rawline in buf[:end + 1].split(b"\n"):
        if not rawline.strip():
            continue
        line_index += 1
        try:
            obj = json.loads(rawline.decode("utf-8", errors="replace"))
        except json.JSONDecodeError:
            print(f"  WARN: {path.name}: Zeile {line_index} kein JSON, "
                  f"uebersprungen (v0-Limitation)")
            continue
        if not isinstance(obj, dict):
            print(f"  WARN: {path.name}: Zeile {line_index} kein Objekt, "
                  f"uebersprungen")
            continue
        parsed.append((line_index, obj))
    return parsed, offset + end + 1, line_index
```

```python
"""HTTP-Client (stdlib urllib). Agent-Dependencies bleiben Python + cryptography."""
import json
from urllib import error, request


class ServerError(Exception):
    pass


def _post(url, payload, token=None, timeout=30):
    data = json.dumps(payload, ensure_ascii=False).encode("utf-8")
    req = request.Request(url, data=data, method="POST",
                          headers={"Content-Type": "application/json"})
    if token:
        req.add_header("Authorization", f"Bearer {token}")
    try:
        with request.urlopen(req, timeout=timeout) as resp:
            return resp.status, json.loads(resp.read().decode("utf-8"))
    except error.HTTPError as e:
        body = e.read().decode("utf-8", errors="replace")
        raise ServerError(f"HTTP {e.code} {url}: {body[:300]}") from e
    except error.URLError as e:
        raise ServerError(f"unreachable {url}: {e.reason}") from e


def enroll(server_url, enrollment_token, pubkey_b64, machine, os_user,
           source_label):
    _, body = _post(f"{server_url}/v1/enroll", {
        "enrollment_token": enrollment_token, "pubkey": pubkey_b64,
        "machine": machine, "os_user": os_user,
        "source_label": source_label})
    return body


def ingest(server_url, token, records):
    _, body = _post(f"{server_url}/v1/ingest", {"records": records},
                    token=token)
    return body["results"]
```

```python
def cmd_scan(args) -> int:
    import hashlib
    import json as _json
    from datetime import datetime, timezone

    from . import scanner, state as st
    from .config import identity, load_config, load_private_key
    from .wrapper import (build_sidecar_wrapper, build_wrapper,
                          extract_session_id)

    try:
        cfg = load_config(args.config)
    except FileNotFoundError:
        print(f"Config nicht gefunden: {args.config} (erst enroll)",
              file=sys.stderr)
        return 3
    priv = load_private_key(cfg.key_path)
    ident = identity(cfg)
    conn = st.open_state(cfg.state_path)

    totals = {"accepted": 0, "duplicate": 0, "rejected": 0}
    rejected_details = []

    for path in scanner.discover_files(cfg.projects_dirs):
        is_subagent = "/subagents/" in str(path)
        file_stem = path.stem
        captured_at = datetime.now(timezone.utc).isoformat()
        wrappers = []

        sidecar_path = path.with_suffix(".meta.json") if is_subagent else None
        sidecar_sha = None
        if sidecar_path is not None and sidecar_path.exists():
            content = sidecar_path.read_text(encoding="utf-8",
                                             errors="replace")
            sidecar_sha = hashlib.sha256(content.encode("utf-8")).hexdigest()
            if not st.sidecar_seen(conn, sidecar_path, sidecar_sha):
                try:
                    sidecar_json = _json.loads(content)
                except _json.JSONDecodeError:
                    sidecar_json = {"unparsed": content[:2000]}
                wrappers.append(build_sidecar_wrapper(
                    sidecar_json, file_name=sidecar_path.name,
                    agent_file_stem=file_stem, captured_at=captured_at,
                    identity=ident, priv_raw=priv))

        parsed, new_offset, new_line_index = scanner.read_new_lines(conn, path)
        for line_index, raw in parsed:
            session_id = extract_session_id(raw, file_stem, is_subagent)
            wrappers.append(build_wrapper(
                raw, session_id=session_id, line_index=line_index,
                captured_at=captured_at, identity=ident, priv_raw=priv))

        if not wrappers:
            continue

        try:
            results = []
            for i in range(0, len(wrappers), 500):
                results.extend(client.ingest(
                    cfg.server_url, cfg.transport_token,
                    wrappers[i:i + 500]))
        except client.ServerError as e:
            print(f"SEND FEHLGESCHLAGEN ({path.name}): {e}", file=sys.stderr)
            print("State unveraendert; naechster Lauf sendet erneut "
                  "(Server-Dedup faengt Doppelte).", file=sys.stderr)
            return 1

        for res in results:
            totals[res["status"]] = totals.get(res["status"], 0) + 1
            if res["status"] == "rejected":
                rejected_details.append(
                    f"{path.name}: {res['record_uuid']} -> "
                    f"{res.get('reason', '?')}")

        stat_ = path.stat()
        st.set_file_state(conn, path, stat_.st_ino, stat_.st_size,
                          new_offset, new_line_index)
        if sidecar_path is not None and sidecar_sha is not None:
            st.mark_sidecar(conn, sidecar_path, sidecar_sha)

    print(f"scan: accepted={totals['accepted']} "
          f"duplicate={totals['duplicate']} rejected={totals['rejected']}")
    if rejected_details:
        for d in rejected_details:
            print(f"  REJECTED {d}", file=sys.stderr)
        return 2
    return 0
```
